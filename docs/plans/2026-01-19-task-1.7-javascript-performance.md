## Issues Found and Fixed

### 1. `lib/item-utils.ts:254-277` - Use toSorted() (js-tosorted-immutable)

**Before:**

```typescript
export function sortItems(items, sortBy) {
  const sorted = [...items];
  switch (sortBy) {
    case "custom":
      return sorted.sort((a, b) => a.order - b.order);
    // ...
  }
}
```

**After:**

```typescript
export function sortItems(items, sortBy) {
  switch (sortBy) {
    case "custom":
      return items.toSorted((a, b) => a.order - b.order);
    // ...
  }
}
```

**Impact**: Cleaner immutable pattern using ES2023 toSorted() instead of spread + sort.

### 2. `lib/progress-utils.ts:67-72` - Combine iterations (js-combine-iterations)

**Before:**

```typescript
const itemsWithMedia = items.filter((item) => item.hasPrimaryMedia).length;
const watchedItems = items.filter(
  (item) =>
    item.hasPrimaryMedia &&
    isFileComplete(item.primaryMediaPosition, item.primaryMediaDuration)
).length;
```

**After:**

```typescript
let itemsWithMedia = 0;
let watchedItems = 0;
for (const item of items) {
  if (item.hasPrimaryMedia) {
    itemsWithMedia++;
    if (isFileComplete(item.primaryMediaPosition, item.primaryMediaDuration)) {
      watchedItems++;
    }
  }
}
```

**Impact**: Single pass through the array instead of two separate filter operations.

### 3. `lib/file-type-utils.ts:85,88,91,128` - Use Set for O(1) lookups (js-set-map-lookups)

**Before:**

```typescript
if ((MEDIA_EXTENSIONS as readonly string[]).includes(ext)) {
  return "MEDIA";
}
// Similar for ARTWORK and SUBTITLE
// ...
if (["srt", "vtt", "sub", "ass"].includes(ext || "")) {
  return "SUBTITLE";
}
```

**After:**

```typescript
// Module-level Sets for O(1) lookups
const MEDIA_EXTENSIONS_SET = new Set<string>(MEDIA_EXTENSIONS);
const ARTWORK_EXTENSIONS_SET = new Set<string>(ARTWORK_EXTENSIONS);
const SUBTITLE_EXTENSIONS_SET = new Set<string>(SUBTITLE_EXTENSIONS);
const SUBTITLE_BARE_SET = new Set(["srt", "vtt", "sub", "ass"]);

// Usage
if (MEDIA_EXTENSIONS_SET.has(ext)) {
  return "MEDIA";
}
```

**Impact**: O(1) lookups instead of O(n) array scans. Especially beneficial for frequently-called file type detection.
